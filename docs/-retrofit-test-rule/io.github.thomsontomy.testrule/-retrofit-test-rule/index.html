RetrofitTestRule

class RetrofitTestRule<T : Any> @JvmOverloads constructor(serviceClass: Class<T>, logLevel: LogLevel = LogLevel.NONE, converterFactories: List<Converter.Factory> = defaultConverterFactories(), callAdapterFactories: List<CallAdapter.Factory> = emptyList(), logger: TestLogger = TestLogger.STDOUT) : TestRule, RetrofitTestDsl<T>

JUnit 4 TestRule that runs a Retrofit service against an OkHttp MockWebServer.

The rule starts the server before the test body, enqueues the responses declared with RestRequest (or programmatically through the RetrofitTestDsl methods), and validates every captured request after the body completes.

class GitHubServiceTest {
    @get:Rule
    val rule = RetrofitTestRule(GitHubService::class.java)

    @Test
    @RestRequest(
        expect = ExpectRequest(path = "/users/me", method = HttpMethod.GET),
        respond = RespondWith(bodyText = """{"login":"octocat"}"""),
    )
    fun `loads the profile`() = runTest {
        val service = rule.getService()
        assertEquals("octocat", service.profile().login)
    }
}
